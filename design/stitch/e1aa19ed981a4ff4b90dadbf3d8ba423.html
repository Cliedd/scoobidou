import asyncio
from passportly import AsyncPassportlyClient

async def check_mobility():
    client = AsyncPassportlyClient(api_key="pk_live_your_key")
    
    # Audit en temps réel des exigences consulaires
    requirement = await client.visa.get(
        origin="CMR",
        destination="FRA",
        passport_type="ORDINARY"
    )
    
    print(f"Statut: {requirement.status}")
    print(f"Taux d'acceptation: {requirement.consular_acceptance_rate * 100}%")
    print(f"Délai moyen: {requirement.average_processing_days} jours")

asyncio.run(check_mobility())

package main

import (
    "context"
    "fmt"
    "github.com/passportly/passportly-go"
)

func main() {
    client := passportly.NewClient("pk_live_token")
    req, err := client.Visa.Get(context.Background(), &passportly.VisaParams{
        Origin: "CMR",
        Destination: "FRA",
    })
    if err != nil {
        panic(err)
    }
    fmt.Printf("Exigence: %s, Délai: %d jours\n", req.Protocol.Status, req.ConsularMetrics.AverageProcessingDays)
}
